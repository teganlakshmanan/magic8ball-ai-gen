let previousY = null; let lastShake = 0; let lastTrigger = 0; let peakYChange = 0; function onMotion(event) { const a = event.accelerationIncludingGravity; if (!a || a.y == null) return; const now = Date.now(); const currentY = a.y; if (previousY === null) { previousY = currentY; return; } // Measure only Y-axis change since the previous reading. const yChange = Math.abs(currentY - previousY); previousY = currentY; // Keep the strongest Y movement during a short shake. if (yChange > peakYChange) { peakYChange = yChange; } // Require a substantial Y change, ignoring tiny fluctuations. if (yChange >= 12) { lastShake = now; } // Trigger once per shake, with a cooldown to prevent duplicates. if ( lastShake && now - lastShake < 250 && peakYChange >= 12 && now - lastTrigger > 1200 ) { lastTrigger = now; lastShake = 0; peakYChange = 0; showAnswer(); } // Reset if no significant movement continues. if (now - lastShake > 300) { lastShake = 0; peakYChange = 0; } }
